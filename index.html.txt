    <script>
        // Sostituisci questo URL con il link RAW del tuo file CSV su GitHub
        const csvUrl = 'https://githubusercontent.com';

        // Funzione per generare i badge colorati in base allo stato
        function getStatusBadge(status) {
            const st = status.trim().toLowerCase();
            if (st === 'in corso') return '<span class="px-2.5 py-1 text-xs font-medium rounded-full bg-emerald-500/10 text-emerald-400 border border-emerald-500/20">In Corso</span>';
            if (st === "lista d'attesa") return '<span class="px-2.5 py-1 text-xs font-medium rounded-full bg-amber-500/10 text-amber-400 border border-amber-500/20">Lista d\'attesa</span>';
            if (st === 'non contattabile') return '<span class="px-2.5 py-1 text-xs font-medium rounded-full bg-slate-600/20 text-slate-400 border border-slate-500/20">Non Contattabile</span>';
            return '<span class="px-2.5 py-1 text-xs font-medium rounded-full bg-rose-500/10 text-rose-400 border border-rose-500/20">Scaduta</span>';
        }

        // Legge il CSV e popola la tabella grafica
        Papa.parse(csvUrl, {
            download: true,
            complete: function(results) {
                const rows = results.data;
                const tbody = document.getElementById('crm-table-body');
                
                // Cicla sui record partendo dal fondo per vedere i più recenti in alto
                for (let i = rows.length - 1; i >= 0; i--) {
                    if (rows[i].length < 2) continue; // Salta righe vuote
                    
                    const id = rows[i][0];
                    const nome = rows[i][5] || 'nd';
                    const arrivo = rows[i][6];
                    const partenza = rows[i][7];
                    const alloggio = rows[i][8] || 'nd';
                    const canale = rows[i][14];
                    const stato = rows[i][21] || 'Scaduta';

                    const tr = document.createElement('tr');
                    tr.className = 'hover:bg-slate-700/30 transition-colors duration-150';
                    tr.innerHTML = `
                        <td class="p-4 font-mono text-xs text-slate-400">${id}</td>
                        <td class="p-4 font-semibold text-white">${nome}</td>
                        <td class="p-4 text-slate-300">${arrivo} - ${partenza}</td>
                        <td class="p-4 text-indigo-300 font-medium">${alloggio}</td>
                        <td class="p-4 text-slate-400">${canale}</td>
                        <td class="p-4">${getStatusBadge(stato)}</td>
                    `;
                    tbody.appendChild(tr);
                }
            }
        });
    </script>
